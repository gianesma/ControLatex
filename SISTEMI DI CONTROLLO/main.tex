\section*{Modellizzazione nel dominio di Laplace}
\[  \begin{cases}
    x(s) = \left(sI-A \right)^{-1}\left( B\ u(s)+P\ z(s)  \right) \\

\section*{Test di Hautus}
Sia $A \in \mathbb{R}^n$ e $\overline{\lambda}$ autovalore di A:
\subsection{Per la raggiungibilità}
\[
rgo(A-\overline{\lambda}I\quad \vdots \quad B) 
\rightarrow 
\begin{cases}
= n \Rightarrow \overline{\lambda} \text{ è raggiungibile}\\
< n \Rightarrow \overline{\lambda} \text{ non è raggiungibile}
\end{cases}
\]

\subsection{Per l'osservabilità}
\[
rgo
\begin{pmatrix}
A-\overline{\lambda }I \\
C    
\end{pmatrix}
\rightarrow
\begin{cases}
= n \Rightarrow \overline{\lambda} \text{ è osservabile}\\
< n \Rightarrow \overline{\lambda} \text{ non è osservabile}
\end{cases}
\]

\section*{Luogo delle radici}                   
Grafico che descrive nel piano complesso come camminano le $n$ radici del polinomio al denominatore al variare del parametro $k$, da $-\infty$ a $\infty$. Si parte dalla seguente struttura:
\[  \hat{F} =  k\cdot \ \frac{\displaystyle\prod_{i=1}^{m}(\displaystyle s-z_i)}{\displaystyle\prod_{i=1}^{n}(\displaystyle s-p_i)}   \]
dove:
\begin{itemize}
    \item $\bm{m}$: numero di zeri
    \item $\bm{n}$: numero di poli
\end{itemize}

\subsection{Centro asintotico}
\[  C.A. = \frac{\displaystyle \sum_{i = 1}^{n} \ p_i- \sum_{i = 1}^{m}\ z_i}{n-m}\]

\subsection{Luogo positivo}
\begin{itemize}
    \item \textbf{Partenza}: $n$ cammini, da ogni polo parte uno e un solo cammino
    \item \textbf{Arrivo}: ogni cammino termina o in uno zero o in un punto asintotico positivo
\end{itemize} 

\subsection{Luogo negativo}
\begin{itemize}
    \item \textbf{Partenza}: $n$ cammini che partono dagli zeri o dai punti asintotici
    \item \textbf{Arrivo}: ogni cammino termina in un polo
\end{itemize}
